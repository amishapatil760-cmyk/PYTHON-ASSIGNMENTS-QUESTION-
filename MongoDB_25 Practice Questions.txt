1. Create/select a database named college:
use college

------

2. Create a collection named students:
db.createCollection("students")

------

3. Display all collections in the current database:
show collections

------

4. Insert one student record into students:
db.students.insertOne({
  name: "Rahul",
  age: 21,
  gender: "Male",
  course: "BCA",
  department: "Computer",
  city: "Pune",
  marks: 80,
  fees: 45000,
  isActive: true
})

------

5. Insert five student records at once:
db.students.insertMany([
  { name: "Amit", age: 22, gender: "Male", course: "BCA", department: "Computer", city: "Pune", marks: 78, fees: 40000, isActive: true },
  { name: "Priya", age: 20, gender: "Female", course: "BSc", department: "IT", city: "Mumbai", marks: 88, fees: 42000, isActive: true },
  { name: "Rohit", age: 23, gender: "Male", course: "BCA", department: "Computer", city: "Nashik", marks: 65, fees: 48000, isActive: false },
  { name: "Sneha", age: 21, gender: "Female", course: "BSc", department: "IT", city: "Pune", marks: 92, fees: 39000, isActive: true },
  { name: "Neha", age: 19, gender: "Female", course: "BCom", department: "Commerce", city: "Nagpur", marks: 72, fees: 35000, isActive: true }
])

------

6. Display all student records:
db.students.find()

------

7. Display only the first five student records:
db.students.find().limit(5)

------

8. Display students whose age is greater than 20:
db.students.find({ age: { $gt: 20 } })

------

9. Display students whose marks are greater than 75:
db.students.find({ marks: { $gt: 75 } })

------

10. Display students who belong to the Computer department:
db.students.find({ department: "Computer" })

------

11. Display students who live in Pune:
db.students.find({ city: "Pune" })

------

12. Display only the name and marks fields of all students:
db.students.find({}, { name: 1, marks: 1, _id: 0 })

------

13. Display students whose gender is Female:
db.students.find({ gender: "Female" })

------

14. Display students whose fees are less than 50000:
db.students.find({ fees: { $lt: 50000 } })

------

15. Display students whose isActive field is true:
db.students.find({ isActive: true })

------

16. Display students whose marks are between 60 and 90 (inclusive):
db.students.find({ marks: { $gte: 60, $lte: 90 } })

------

17. Display students who belong to either the Computer or IT department:
db.students.find({ department: { $in: ["Computer", "IT"] } })

------

18. Display students whose age is not equal to 20:
db.students.find({ age: { $ne: 20 } })

------

19. Sort students by marks in ascending order:
db.students.find().sort({ marks: 1 })

------

20. Sort students by marks in descending order:
db.students.find().sort({ marks: -1 })

------

21. Update the marks of the student named Rahul to 85:
db.students.updateOne(
  { name: "Rahul" },
  { $set: { marks: 85 } }
)

------

22. Increase the fees of the student named Amit by 5000:
db.students.updateOne(
  { name: "Amit" },
  { $inc: { fees: 5000 } }
)

------

23. Add a new field named email to the student named Rahul:
db.students.updateOne(
  { name: "Rahul" },
  { $set: { email: "rahul@example.com" } }
)

------

24. Delete one student whose name is Rohit:
db.students.deleteOne({ name: "Rohit" })

------

25. Delete all student records whose marks are less than 35:
db.students.deleteMany({ marks: { $lt: 35 } })
